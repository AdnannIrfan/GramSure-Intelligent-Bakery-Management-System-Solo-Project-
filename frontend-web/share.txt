import { useEffect, useState } from 'react';
import apiClient from '../../services/apiClient';
import Badge from '../../components/common/Badge';

export default function BranchesPage() {
  const [branches, setBranches] = useState([]);
  const [dashboard, setDashboard] = useState(null);
  const [loading, setLoading] = useState(true);
  const [showForm, setShowForm] = useState(false);
  const [form, setForm] = useState({ name: '', address: '' });

  useEffect(() => {
    loadData();
  }, []);

  async function loadData() {
    setLoading(true);
    try {
      const [branchesRes, dashboardRes] = await Promise.all([
        apiClient.get('/branches'),
        apiClient.get('/branches/consolidated-dashboard'),
      ]);
      setBranches(branchesRes.data.data);
      setDashboard(dashboardRes.data.data);
    } catch (err) {
      console.error('Failed to load branch data:', err);
    } finally {
      setLoading(false);
    }
  }

  async function handleSubmit(e) {
    e.preventDefault();
    try {
      await apiClient.post('/branches', form);
      setShowForm(false);
      setForm({ name: '', address: '' });
      loadData();
    } catch (err) {
      alert(err.response?.data?.message || 'Failed to create branch');
    }
  }

  const inputClass =
    'border border-gray-200 rounded-xl px-3.5 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-gramsure-gold focus:border-transparent transition-shadow placeholder:text-gray-400';

  return (
    <div className="min-h-screen bg-gray-50">
      <div className="bg-gradient-to-r from-gramsure-navy to-slate-800 px-6 pt-8 pb-14">
        <div className="max-w-[1400px] mx-auto flex items-center justify-between">
          <div>
            <h1 className="text-2xl font-bold text-white">Multi-Branch & Franchise Management</h1>
            <p className="text-sm text-white/50 mt-1">{branches.length} branch{branches.length !== 1 ? 'es' : ''}</p>
          </div>
          <button
            onClick={() => setShowForm(!showForm)}
            className="bg-gramsure-gold text-gramsure-navy px-4 py-2.5 rounded-xl text-sm font-semibold hover:opacity-90 active:scale-[0.98] transition-all shadow-lg shadow-black/10"
          >
            {showForm ? 'Cancel' : '+ New Branch'}
          </button>
        </div>
      </div>

      <div className="max-w-[1400px] mx-auto px-6 -mt-8 pb-10 space-y-6">
        {showForm && (
          <form
            onSubmit={handleSubmit}
            className="bg-white rounded-2xl shadow-sm border border-gray-100 p-5 grid grid-cols-1 sm:grid-cols-2 gap-4"
          >
            <div className="sm:col-span-2">
              <h3 className="text-sm font-semibold text-gray-800 mb-1">New Branch Details</h3>
            </div>
            <input
              placeholder="Branch Name"
              value={form.name}
              onChange={(e) => setForm({ ...form, name: e.target.value })}
              required
              className={inputClass}
            />
            <input
              placeholder="Address"
              value={form.address}
              onChange={(e) => setForm({ ...form, address: e.target.value })}
              className={inputClass}
            />
            <button
              type="submit"
              className="sm:col-span-2 bg-gramsure-navy text-white font-semibold py-2.5 rounded-xl text-sm hover:opacity-90 active:scale-[0.99] transition-all"
            >
              Save Branch
            </button>
          </form>
        )}

        {loading ? (
          <div className="bg-white rounded-2xl shadow-sm border border-gray-100 p-6 space-y-3">
            {[...Array(3)].map((_, i) => (
              <div key={i} className="h-10 bg-gray-100 rounded-lg animate-pulse" />

          </div>
        ) : (
          <>
            {dashboard?.underperformingBranches?.length > 0 && (
              <div className="bg-[#f97316]/5 border border-[#f97316]/20 rounded-2xl p-5">
                <div className="flex items-center gap-2 mb-3">
                  <span className="w-6 h-6 rounded-lg bg-[#f97316]/10 flex items-center justify-center text-[#f97316] text-xs font-bold shrink-0">
                    !

                    <li key={b.branchId} className="list-disc">
                      {b.branchName} â€” {b.percentBelowAverage}% below average revenue
                    </li>
                  ))}
                </ul>
              </div>

            <div className="bg-white rounded-2xl shadow-sm border border-gray-100 overflow-hidden">

                <thead className="bg-gray-50/80 text-gray-500 text-xs uppercase tracking-wide">

                    <th className="text-left px-5 py-3 font-semibold">Branch</th>

                    <th className="text-left px-5 py-3 font-semibold">30-day Revenue</th>

===== services\apiClient.js =====

import axios from 'axios';

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:5000/api';

const apiClient = axios.create({
  baseURL: API_BASE_URL,
  headers: { 'Content-Type': 'application/json' },
});


apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('gramsure_access_token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});


let isRefreshing = false;
let refreshQueue = [];

apiClient.interceptors.response.use(
  (response) => response,
  async (error) => {
    const originalRequest = error.config;

    if (error.response?.status === 401 && !originalRequest._retry) {
      if (isRefreshing) {
        return new Promise((resolve, reject) => {
          refreshQueue.push({ resolve, reject });
        }).then((token) => {
          originalRequest.headers.Authorization = `Bearer ${token}`;
          return apiClient(originalRequest);
        });
      }

      originalRequest._retry = true;
      isRefreshing = true;

      const refreshToken = localStorage.getItem('gramsure_refresh_token');
      if (!refreshToken) {
        localStorage.clear();
        window.location.href = '/login';
        return Promise.reject(error);
      }

      try {
        const { data } = await axios.post(`${API_BASE_URL}/auth/refresh`, { refreshToken });
        const newAccessToken = data.data.accessToken;
        localStorage.setItem('gramsure_access_token', newAccessToken);

        refreshQueue.forEach(({ resolve }) => resolve(newAccessToken));
        refreshQueue = [];

        originalRequest.headers.Authorization = `Bearer ${newAccessToken}`;
        return apiClient(originalRequest);
      } catch (refreshError) {
        refreshQueue.forEach(({ reject }) => reject(refreshError));
        refreshQueue = [];
        localStorage.clear();
        window.location.href = '/login';
        return Promise.reject(refreshError);
      } finally {
        isRefreshing = false;
      }
    }

    return Promise.reject(error);
  }
);

export default apiClient;

function ProtectedRoute({ children }) {
  const { isAuthenticated, loading } = useAuth();
  if (loading) return <div className="flex items-center justify-center h-screen">Loading...</div>;
  if (!isAuthenticated) return <Navigate to="/login" replace />;
  return children;
}

===== context\AuthContext.jsx =====

import { createContext, useState, useEffect } from 'react';
import apiClient from '../services/apiClient';

export const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadUserFromToken();
  }, []);

  async function loadUserFromToken() {
    const token = localStorage.getItem('gramsure_access_token');
    if (!token) {
      setLoading(false);
      return;
    }

    try {
      const { data } = await apiClient.get('/auth/me');
      setUser(data.data);
    } catch (err) {
      localStorage.clear();
      setUser(null);
    } finally {
      setLoading(false);
    }
  }

  async function login(phone, password) {
    const { data } = await apiClient.post('/auth/login', { phone, password });
    const { user: loggedInUser, accessToken, refreshToken } = data.data;

    localStorage.setItem('gramsure_access_token', accessToken);
    localStorage.setItem('gramsure_refresh_token', refreshToken);
    setUser(loggedInUser);

    return loggedInUser;
  }

  function logout() {
    localStorage.clear();
    setUser(null);
  }

  const value = { user, loading, login, logout, isAuthenticated: !!user };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}
